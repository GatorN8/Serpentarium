async function loadLive(force=false){

if(ui.atlasMode!=='live')return;

let key=live.query.toLowerCase().trim()+':'+live.page;


if(!force&&live.cache.has(key)){

let c=live.cache.get(key);

live.results=c.results;

live.total=c.total;

live.received=true;

live.error='';

live.loading=false;

render();

return;

}


if(live.request)live.request.abort();

const controller=new AbortController();

live.request=controller;

const serial=++live.serial;

live.loading=true;

live.error='';

live.results=[];

render();


try{

const gap=1100-(Date.now()-live.lastRequest);

if(gap>0)await new Promise(r=>setTimeout(r,gap));

if(serial!==live.serial)return;

live.lastRequest=Date.now();

const search=live.query.trim().length>0;

const url=search

?'https://api.inaturalist.org/v1/taxa/autocomplete?q='+encodeURIComponent(live.query.trim())+'&per_page=100'

:'https://api.inaturalist.org/v1/observations/species_counts?taxon_id=85553&per_page=36&page='+live.page;


const timeout=setTimeout(()=>controller.abort(),13000);

let res;

try{

res=await fetch(url,{signal:controller.signal,headers:{Accept:'application/json'}});

}finally{

clearTimeout(timeout);

}


if(!res.ok)throw new Error(res.status===429?'Too many requests; please retry later.':'Service returned HTTP '+res.status);


const data=await res.json();

if(!data||!Array.isArray(data.results))throw new Error('Unexpected response from species service.');


let entries=data.results.map(rec=>normalizeLive(rec,search)).filter(Boolean);

const uniq=new Set();

entries=entries.filter(t=>!uniq.has(t.id)&&uniq.add(t.id));


if(serial!==live.serial)return;

live.results=entries;

live.total=Number(data.total_results)||0;

live.received=true;

live.cache.set(key,{results:entries,total:live.total});

}catch(e){

if(serial!==live.serial)return;

live.error=e.name==='AbortError'?'Timed out or interrupted. Please retry.':''+(e.message||e);

live.received=false;

}finally{

if(serial===live.serial){

live.loading=false;

live.request=null;

render();

}

}

}

function quizOptions(s,idx){

let others=SPECIES.filter(v=>v.id!==s.id)

.sort((a,b)=>((a.name.charCodeAt(0)+idx*11)%37)-((b.name.charCodeAt(0)+idx*11)%37))

.slice(0,3);


return [s,...others].sort((a,b)=>((a.name.length*13+idx*7)%17)-((b.name.length*13+idx*7)%17));

}

function validateBackup(input){

if(!input||input.app!=='Serpentarium'||input.schema!==1||!input.saved||input.saved.version!==1){

throw Error('Not a Serpentarium v1 backup');

}


let s=input.saved;

if(

!Array.isArray(s.researched)||

!Array.isArray(s.favorites)||

!Array.isArray(s.claimed)||

!Array.isArray(s.daily)||

!Array.isArray(s.morphs)||

!s.habitats||

typeof s.habitats!=='object'||

typeof s.points!=='number'||

!Number.isFinite(s.points)||

s.points<0||

s.points>1e7||

s.researched.length>2000||

s.claimed.length>100000

){

throw Error('Backup data is invalid');

}


let ids=new Set(SPECIES.map(s=>s.id));


for(let id of [...s.researched,...s.favorites]){

if(!ids.has(id))throw Error('Unknown species in backup');

}


for(let id of s.morphs){

if(!MORPHS.some(m=>m.id===id))throw Error('Unknown morph in backup');

}


for(let [id,h] of Object.entries(s.habitats)){

let def=HABITATS.find(x=>x.id===id);

if(

!def||

!h||

!Number.isInteger(h.level)||

h.level<1||

h.level>4||

!Array.isArray(h.exhibits)||

h.exhibits.some(v=>!def.species.includes(v)||!ids.has(v))

){

throw Error('Invalid sanctuary state');

}

}


if(s.liveBookmarks!==undefined){

if(

!Array.isArray(s.liveBookmarks)||

s.liveBookmarks.length>400||

s.liveBookmarks.some(t=>

!t||

!Number.isInteger(t.id)||

t.id<=0||

typeof t.latin!=='string'||

t.latin.length>160||

typeof t.name!=='string'||

t.name.length>180

)

){

throw Error('Invalid online bookmarks in backup');

}

}


return {

...base(),

...s,

liveBookmarks:Array.isArray(s.liveBookmarks)?s.liveBookmarks:[]

};

}

if(!f||f.size>5e6){

try{

let p=JSON.parse(await f.text());

let r=validateBackup(p);

if(!confirm('Replace the Serpentarium progress saved on this device with this backup?'))return;

state=r;

});
